🔍 Search Videos

Search for YouTube videos, channels, or playlists

GET /youtube/search?query={query}&limit={limit}&type={type} 
// Example
fetch('/youtube/search?query=javascript+tutorial&limit=10&type=video')
  .then(res => res.json())
  .then(data => console.log(data));

📊 Get Video Info

Get detailed information about a YouTube video

GET /youtube/info/{videoId} 
// Example
fetch('/youtube/info/dQw4w9WgXcQ')
  .then(res => res.json())
  .then(data => console.log(data));

⬇️ Download Video

Download a YouTube video

GET /youtube/download/{videoId}?quality={quality}&filter={filter} 
// Example - Download highest quality
window.location.href = '/youtube/download/dQw4w9WgXcQ?quality=highest';

// Example - Download audio only
window.location.href = '/youtube/download/dQw4w9WgXcQ?filter=audioonly';

📺 Stream Video

Stream a YouTube video without downloading

GET /youtube/stream/{videoId}?quality={quality} 
// Example - Use in HTML video player
<video controls>
  <source src="/youtube/stream/dQw4w9WgXcQ" type="video/mp4">
</video>

📋 Get Playlist Info

Get information about a YouTube playlist

GET /youtube/playlist/{playlistId}?limit={limit} 
// Example
fetch('/youtube/playlist/PLrAXtmErZgOeiKm4sgNOknGvNjby9efdf?limit=10')
  .then(res => res.json())
  .then(data => console.log(data));
